\begin{frame}{Why care? Verifying smart contracts}
	A \alert{smart contract} is a program on a blockchain that holds
	and moves cryptocurrency.
	\begin{block}{BeautyChain (BEC) token hack, 2018}
		\small
		\begin{semiverbatim}
			\alert{uint256} amount = uint256(cnt) * \_value;
		\end{semiverbatim}
		\normalsize
		$\begin{array}{l}
				\left.
				\begin{array}{l}
					\texttt{cnt}\text{: \# recipients} \\
					\texttt{\_value}\text{: tokens sent to each}
				\end{array}
				\right\}\ \alert{\text{attacker's pick}} \\
				\hphantom{x}\texttt{amount}\text{: what the attacker pays the bank}
			\end{array}$
		\[
			\texttt{cnt}=2,\ \texttt{\_value}=2^{255}
			\Rightarrow \alert{\text{\texttt{amount}}=0}
			\Rightarrow \text{free tokens!}
		\]
	\end{block}
	\pause
	\begin{itemize}
		\item This is a nonlinear arithmetic question!
	\end{itemize}
\end{frame}

\begin{frame}{Quantifier-Free Nonlinear Integer Arithmetic (NIA)}
	\begin{itemize}
		\item Quantifier-free formulas over the \alert{integers} with
		      $\{+,-,\cdot,\mathit{div},\mathit{mod},\mathit{abs}\}$
		\item Truly nonlinear:
		      \begin{itemize}
			      \item variables multiplied together ($x\cdot y$, $x^2$, $x^ky^l$)
			      \item $\mathit{div}$/$\mathit{mod}$ with a \alert{non-constant divisor}
			            ($\divc{x}{y}$, $\modc{x}{y}$)
		      \end{itemize}
		\item Satisfiability is \alert{undecidable} \\
		      (Hilbert's 10th problem, Matiyasevich)
	\end{itemize}
	\begin{block}{This talk}
		An incomplete but practical decision procedure for NIA
	\end{block}
\end{frame}

\begin{frame}{Simple-looking equations defeat state-of-the-art solvers}
	Find integers $x,\;y,\;z$ such that:
	\[
		x^3 + y^3 + z^3 = 79
	\]
	\begin{itemize}
		\item A ``sum of three cubes'' problem
		\item A job for \alert{SMT solvers} $\leftarrow$ SAT solving + First order theories
		\item cvc5, MathSAT, Yices~2 and Z3 all fail
		      \pause
		\item Our solver finds $x = -19,\; y = 35,\; z = -33$ in $\sim$20s
	\end{itemize}
\end{frame}

\begin{frame}{Can we get away with just LIA?}
	\begin{itemize}
		\item \alert{LIA} (linear arithmetic)
		      \begin{itemize}
			      \item easy for SMT solvers
			      \item Simplex algorithm + branch-and-bound
		      \end{itemize}
		\item \alert{NIA} (nonlinear, current problem)
		      \begin{itemize}
			      \item even small problems don't get solved
		      \end{itemize}
	\end{itemize}
	\begin{block}{Idea}
		Can we approximate NIA problems using LIA?
	\end{block}
	\pause
	Yes, by \alert{linearization}: approximate nonlinear terms with linear
	axioms, then refine.
\end{frame}

\begin{frame}{The idea, on an example}
	\[
		\varphi:\ \alert{x^2}\leq x \land x\geq 2
	\]
	\pause
	Replace $x^2$ by a fresh constant $p$ (a \alert{pure}).
	We get a LIA formula:
	\[
		\widehat\varphi:\ \alert{p}\leq x \land x\geq 2
	\]
	Add a refinement axiom, a \alert{linearization} of $x^2$:
	\[
		\widehat\varphi \;\land\; (x\geq 2\Rightarrow \alert{p}\geq 2x)
	\]
	\pause
	This forces:
	\[
		2x \leq \alert{p} \leq x \;\models\; x\leq 0
	\]
	unsat
\end{frame}

\section{Algorithm}

\begin{frame}{Normalization and purification}
	\begin{block}{Normalization}
		Every nonlinear monomial is rewritten into the form $c\cdot x^ky^l$.
		\begin{itemize}
			\item $c$ \quad concrete integer
			\item $x, y$ \quad variables
		\end{itemize}
	\end{block}
	\begin{block}{Purification}
		Every nonlinear subterm $t$ is replaced by a fresh constant $\pure{t}$.
		Call $\pure{t}$ a \alert{pure}.
	\end{block}
	\begin{itemize}
		\item mul: \quad $x^ky^l \;\rightsquigarrow\; \pure{x^ky^l}$
		\item div: \quad $\divc{x}{y} \;\rightsquigarrow\; \pure{\divc{x}{y}}$
		\item mod: \quad $\modc{x}{y} \;\rightsquigarrow\; \pure{\modc{x}{y}}$
	\end{itemize}
	\begin{block}{Result}
		A LIA formula $\widehat\varphi$ over the original variables and pures.
	\end{block}
\end{frame}

	\begin{itemize}
		\item $\widehat\varphi \gets \textsc{Purify}(\varphi)$ is an
		      \alert{overapproximation} of $\varphi$ $\Rightarrow$
		      \textsc{unsat} transfers

		      be \alert{spurious} (next slide)
	\end{itemize}

\begin{frame}{Checking $\mu$ against NIA semantics}
	\begin{block}{\textsc{Check}($\varphi,\widehat\varphi,P,\mu,\mathcal{A}$)}
		\small
		\begin{enumerate}
			\item \textbf{if} $\mu\models_{\mathrm{NIA}}\varphi$ \textbf{then return}
			      $\emptyset$
			      \hfill\emph{(values of pures irrelevant)}
			      \pause
			\item $I\gets\textsc{Implicant}(\widehat\varphi,\mu)$
			      \hfill\emph{($I \models \widehat\varphi$)}
			      \pause
			\item $R\gets\{\pure{t}\in I \mid \mu(\pure{t})\neq\mu(t)\}$
			      \hfill\emph{(collect wrong pures)}
			      \begin{itemize}
				      \item For example: $p \approx x^3$, $\mu(p) = 5$, $\mu(x) = 2$
			      \end{itemize}
			      \pause
			\item $\mathit{res},\mathcal{N}\gets\textsc{Model-Fix}(\mu,R)$
			      \hfill\emph{(is $\mu$ \textbf{almost} correct?)}
			\item \textbf{if} $\mathit{res}$ \textbf{then return} $\emptyset$
			      \pause
			\item \textbf{for each} $p\in R$:\quad
			      $\mathcal{N}\gets\mathcal{N}\cup\textsc{Axioms}(p,\mu)$
			\item \textbf{return} $\mathcal{N}$
		\end{enumerate}
	\end{block}
\end{frame}

\begin{frame}{Model-Fix}
	Adding axioms makes $\widehat\varphi$ harder to solve. Instead, run cheap
	``sub-iterations'' on the spurious model.
	\pause
	\begin{itemize}
		\item Pin variables irrelevant to the failing literals
		\item Sub-iterate on the restricted problem
		\item NIA-valid model appears $\Rightarrow$ answer \alert{\textsc{sat}} directly
		\item Otherwise, keep the gathered axioms and go back\\ to the main loop
	\end{itemize}
\end{frame}

\begin{frame}{Same example, more verbose}
	\[
		\varphi:\quad \alert<1>{x^2}\leq x \;\land\; x\geq 2
	\]
	\begin{enumerate}
		\small
		\item<1-> Purify: $\alert<1>{p}:=\pure{x^2}$
		      \[
			      \hspace*{-\leftmargin}\makebox[\textwidth][c]{$\widehat\varphi:\quad
					      \alert<1>{p}\leq x\land x\geq 2$}
		      \]
		\item<2-> LIA solve: $\mu=\{x\mapsto 2,\ p\mapsto 1\}$
		\item<3-> NIA check: $\mu(x^2) \neq \mu(p) \Rightarrow \alert{\text{$\mu$ is spurious}}$\\
		      $I=\{p\leq x,\ x\geq 2\}$, $R=\{p\}$
		\item<4-> add axioms for $p$, among them \textit{Linear Lower Bound}:
		      \[
			      \hspace*{-\leftmargin}\makebox[\textwidth][c]{$x\geq 2 \;\Rightarrow\; p\geq 2x$}
		      \]
		\item<5-> LIA solve: $2x\leq p\leq x$ gives $x\leq 0$, contradicting
		      $x\geq 2$
		\item<6-> the abstraction is unsat $\Rightarrow$ $\varphi$ is
		      \alert{\textsc{unsat}}
	\end{enumerate}
\end{frame}

\section{Refinement Axioms}

\begin{frame}{Quick reminder}
	\begin{itemize}
		\item $\widehat\varphi\land\bigwedge\mathcal{A}$ is \textbf{sat}, but for
		      some nonlinear term $t$:
		      \[
			      \mu(\pure{t})\;\neq\;\mu(t)
		      \]
		\item The model is \alert{spurious} $\Rightarrow$ it satisfies $\widehat\varphi$,
		      not $\varphi$
		      \vskip 1cm
		\item Goal: linear axioms that make $\pure{t}$ track $t$ more closely and \alert{rule out}
		      this bad model (or more models)
		\item Quality of the axioms decides everything:
		      \begin{itemize}
			      \item \emph{tight} axioms cut iterations
			      \item \emph{cheap} axioms keep each iteration fast
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Axioms for $\pure{x^k}$}
	\begin{center}
		\sbox0{%
			\begin{tabular}{ll}
				\toprule
				\textit{Sign}    & $\pure{x^k}\geq 0$ \ ($k$ even), \quad
				$x>0\iff\pure{x^k}>0$ \ ($k$ odd)                                                   \\
				\textit{Eq-zero} & $(\pure{x^k}=0)\iff(x=0)$                                       \\
				\textit{Gap}     & $\pure{x^k}\leq v^k \lor \pure{x^k}\geq(v+1)^k$                 \\
				\textit{Mod}     & $x\equiv v\!\!\pmod m \implies \pure{x^k}\equiv v^k\!\!\pmod m$ \\
				\bottomrule
			\end{tabular}}%
		\usebox0

		\begin{tabular*}{\wd0}{ll@{\extracolsep{\fill}}}
			\toprule
			\textit{Linear lower bound} & $x\geq v \implies \pure{x^k}\geq v^{k-1}x$        \\
			\textit{Linear upper bound} & $0\leq x\leq v \implies \pure{x^k}\leq v^{k-1}x$ \\
			\bottomrule
		\end{tabular*}
	\end{center}
\end{frame}

\begin{frame}{Bounding mixed powers $\pure{x^ky^l}$}
	\textbf{Goal:} keep the bound \emph{linear} in the variables $\pure{x^my^n}$ depends on
	\begin{itemize}
		\item Linearize $x^m$ via the secant bound; treat $y^n$ constant
		\item e.g.
		      \[
			      x\geq 3\land y\geq 5\Rightarrow\pure{x^3y^2}\geq 3^2 \,5^2 \,x
		      \]
		      \[
			      0\leq x\leq 3\land 0\leq y\leq 5\Rightarrow\pure{x^3y^2}\leq 3^2 \,5^2 \,x
		      \]
		\item Symmetric construction: fix $x^m$ constant, linearize $y^n$
		      instead
	\end{itemize}
\end{frame}

\section{Experiments}

\begin{frame}{Setup}
	\begin{itemize}
		\item \alert{\solver} implemented in C++, using Z3 as the LIA backend
		\item Full SMT-LIB NIA set $\rightarrow$ 25\,444 instances
		      % \item 180\,s wall clock
		      % \item Baselines: Z3~4.15.4, MathSAT5~5.6.16, Yices~2.6.4, cvc5~1.3.2
		\item Tested against: Z3, MathSAT5, Yices, cvc5
		\item Four configurations:
		      \begin{itemize}
			      \small
			      \item base (as presented here)
			      \item {+}\textsc{Frontier}: reduces the overhead of tangent plane axioms
			      \item {+}\textsc{Congr.}: pures with equal arguments are equal
			      \item {+}\textsc{Congr.}\,$-$\,\textsc{Model-Fix}: congruence on, model-fix off
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Conclusions and future work}
	\begin{block}{Summary}
		\begin{itemize}
			\item NIA $\to$ LIA by purification into plain constants
			\item new \alert{secant-based} bounds for $x^k$ and $x^ky^l$
			\item competitive overall, \alert{dominant} on polynomial benchmarks
		\end{itemize}
	\end{block}
	\begin{block}{Future work}
		\begin{itemize}
			\item part of our project on quantified NIA
			\item bit-blasting preprocessing for small-model sat instances
			\item pruning axioms made redundant by stronger ones
		\end{itemize}
	\end{block}
\end{frame}
